country_event = { # Reign of Jiri z Podebrad - hidden setup
    id = vu_Bohemian_Events.8
    title = vu_Bohemian_Events.8.t
    desc = vu_Bohemian_Events.8.d
    picture = vu_jiri_podebrad_eventPicture
    hidden = yes
    fire_only_once = yes

    # Fires once Jiri has taken the throne (1457) and before he dies (1471).
    # Sets the country flag that enables his succession event (.9).
    trigger = {
        tag = BOH
        is_year = 1457
        NOT = { is_year = 1471 }
        dynasty = "z Podebrad"
        NOT = { has_country_flag = boh_has_jiri_flag }
    }

    mean_time_to_happen = {
        months = 1
    }

    option = {
        name = vu_Bohemian_Events.8.a
        set_country_flag = boh_has_jiri_flag
        set_ruler_flag = jiri_alive_flag
    }
}

country_event = { #The Succession of Jiri z Podebrad
    id = vu_Bohemian_Events.9
    title = vu_Bohemian_Events.9.t
    desc = vu_Bohemian_Events.9.d
    picture =  DEATH_OF_HEIR_eventPicture

    fire_only_once = yes

    major = yes

    mean_time_to_happen = {
        months = 1
    }
    trigger = {
        tag = BOH
        NOT = { has_ruler_flag = jiri_alive_flag }
        has_country_flag = boh_has_jiri_flag
        NOT = { stability = 3 }
        NOT = { is_year = 1496 }
        valid_for_personal_unions_trigger = yes
    }

    option = {
        name = vu_Bohemian_Events.9.a

        add_government_reform = elective_monarchy
        hidden_effect = {
            kill_ruler = yes
            remove_heir = {}
        }
        define_ruler = {
            dynasty = Jagiellon
            change_adm = 1
            change_mil = 1
            age = 37
        }
        if = {
            limit = {
                POL = { valid_for_personal_unions_trigger = yes }
            }
            POL = {
                add_casus_belli = {
                    type = cb_restore_personal_union
                    months = 1200
                    target = ROOT
                }
            }
        }
        if = {
            limit = {
                has_dlc = "Rights of Man"
            }
            hidden_effect = { add_ruler_personality = cruel_personality }
        }
    }

    option = {
        name = vu_Bohemian_Events.9.b

        trigger = {
            HUN = {
                is_subject = no
                NOT = { war_with = ROOT }
                ai = yes #I had to add this...
            }
        }

        HUN = { create_union = ROOT } #Really? no checks for HUN being AI?
        add_liberty_desire = 25

        ai_chance = { factor = 0 }
    }

    option = {
        name = vu_Bohemian_Events.9.c

        add_stability = -1
        add_legitimacy = 15
        if = {
            limit = {
                POL = { valid_for_personal_unions_trigger = yes }
            }
            POL = {
                add_casus_belli = {
                    type = cb_restore_personal_union
                    months = 1200
                    target = ROOT
                }
            }
        }
        if = {
            limit = {
                HUN = { valid_for_personal_unions_trigger = yes }
            }
            HUN = {
                add_casus_belli = {
                    type = cb_restore_personal_union
                    months = 1200
                    target = ROOT
                }
            }
        }
    }
}
