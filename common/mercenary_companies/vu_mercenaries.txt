# ── OTTOMAN-STYLE (hired by Europeans) ──────────────────────────────────────

# The Janissary Corps — elite devshirme infantry of the Sublime Porte.
# Hired by European border nations who have encountered Ottoman methods of war
# firsthand. The Ottomans themselves already field Janissaries as standing troops
# and have no need to hire them as mercenaries.
vu_merc_ottoman_janissaries = {
	regiments_per_development = 0.05
	cavalry_weight = 0
	artillery_weight = 0
	cost_modifier = 0.75		# Foreign mercenaries — expensive but formidable
	home_province = 151		# Istanbul — where these soldiers were trained
	sprites = { muslimgfx_sprite_pack }
	trigger = {
		# Available to European nations that border or contest the Ottoman sphere.
		# NOT available to the Ottomans themselves — they already have Janissaries.
		NOT = { tag = OTT }
		NOT = { tag = TUR }
		NOT = { religion_group = muslim }
		OR = {
			capital_scope = { region = balkan_region }
			capital_scope = { region = carpathia_region }
			capital_scope = { region = austria_region }
			capital_scope = { region = poland_region }
			tag = VEN		# Venice — long trade and conflict with the Porte
		}
	}
	modifier = {
		discipline = 0.05
		infantry_power = 0.10
		shock_damage_received = -0.05
	}
}
