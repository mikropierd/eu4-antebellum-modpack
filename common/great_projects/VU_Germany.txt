				}
			}

			limit = { has_dlc = "Common Sense" }

					has_parliament = yes

		else = {
			FROM = {
				OR = {
					government = republic

				}
			}
		}
	}

		FROM = { adm_tech = 37 }
		FROM = {
			OR = {
				tag = GER
				tag = PRU
				tag = HLR
				AND = {
					culture_group = germanic
					NOT = { primary_culture = dutch }
				}
			}
		}
		if = {
			limit = { has_dlc = "Common Sense" }
			FROM = {
				OR = {
					has_parliament = yes
					government = republic
				}
			}
		}
		else = {
			FROM = {
				OR = {
					government = republic
					has_reform = parliamentary_reform
				}
			}
		}
	}
	keep_trigger = { }
	tier_0 = {
		upgrade_time = { months = 0 }
		cost_to_upgrade = { factor = 0 }
		province_modifiers = { }
		area_modifier = { }
		country_modifiers = { }
		on_upgraded = { }
	}
	tier_1 = {
		upgrade_time = { months = 120 }
		cost_to_upgrade = { factor = 1000 }
		province_modifiers = {
			local_monthly_devastation = -0.05
		}
		area_modifier = { }
		country_modifiers = { 
			governing_capacity = 0.1
		}
		on_upgraded = {
			add_devastation = -10
		}
	}
	tier_2 = {
		upgrade_time = { months = 240 }
		cost_to_upgrade = { factor = 2500 }
		province_modifiers = {
			local_monthly_devastation = -0.1
		}

			local_unrest = -1
		}
		country_modifiers = {

		}
		on_upgraded = { }
