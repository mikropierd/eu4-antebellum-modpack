# VU Province-Count Force Limit Scaling
# Small nations receive a force limit bonus. Large empires receive a
# penalty. The baseline band (51-100 provinces) gets no modifier.
# Thresholds are calibrated for VU's expanded map (~2x vanilla provinces).
# Design: VU
##########################################################################

# 1-10 provinces — Independent Principality
# Minor lords, free cities, OPMs. Disproportionate military capacity
# to survive in a world of larger neighbours.
vu_army_scale_micro = {
	potential = {
		num_of_cities = 1
		NOT = { num_of_cities = 11 }
	}
	trigger = {
		num_of_cities = 1
		NOT = { num_of_cities = 11 }
	}
	land_forcelimit_modifier = 0.20
	naval_forcelimit_modifier = 0.20
}
