		OR = {
			has_mission = ab_russia_bulgaria

				estate_privilege = estate_nobles_balkan_client
			}
		}
	}

		custom_trigger_tooltip = {
			tooltip = has_completed_ab_russia_bulgaria
			has_unlocked_estate_privilege = {
				estate_privilege = estate_nobles_balkan_client
			}
		}
		calc_true_if = {
			amount = 3
			all_subject_country = {
				capital_scope = {
					region = balkan_region
				}

					is_march = yes
					is_vassal = yes
				}
			}
		}
	}
	penalties = {
	}
	benefits = {
		diplomatic_upkeep = 3
	}
	ai_will_do = {
		factor = 1
	}
}
